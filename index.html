import { MongoClient } from 'mongodb'; // Canada Sea Ops — dashboard data route (same contract as the USA / Thailand mongo-batch.js) // GET → { records, users, builtAt } // POST (x-batch-secret) { action:'wipe', direction } → clears ca_ops_export / ca_ops_import // POST (x-batch-secret) { action:'push', direction, records } → inserts a chunk (CanadaOpsPush.gs) // POST (x-batch-secret) { action:'wipe'|'push', collection:'estimatedvsactual', records } // → charge-level ETD/ETA tab (CanadaEstVsActPush.gs); only whitelisted collections allowed const uri = process.env.MONGO_URI; const SECRET = process.env.BATCH_SECRET; const DB = process.env.MONGO_DB || undefined; // undefined → database in the URI (sar-ca-ops) const COLS = { Export: 'ca_ops_export', Import: 'ca_ops_import' }; const EXTRA_COLLECTIONS = new Set(['estimatedvsactual']); // writable by name via POST const TTL_MS = 15 * 60 * 1000; // 15 min (Refresh button bypasses it) // Empty cells are dropped from each record to keep the response well under Vercel's 4.5 MB limit. // These keys are always kept because the dashboard checks for their presence. const ALWAYS_KEEP = new Set(['shipmentId', 'direction', 'lob', 'jobDate', 'periodDate', 'isOpen', 'jobStatus', 'jobOwner', 'unbilledRevenue', 'unpostedCost', 'evaUnbilledRevLines', 'evaUnbilledCostLines']); function compact(doc) { const out = {}; for (const [k, v] of Object.entries(doc)) { if (k === '_id') continue; if (!ALWAYS_KEEP.has(k) && (v === '' || v === null || v === undefined)) continue; out[k] = v; } return out; } let mongoClient; async function getClient() { if (!mongoClient) { mongoClient = new MongoClient(uri, { maxPoolSize: 5, socketTimeoutMS: 45000 }); await mongoClient.connect(); } return mongoClient; } // Unbilled Revenue / Unposted Cost come from the charge-level estimatedvsactual collection: // per job (jobNo = shipmentId) the charge lines' unbilledRevenue / unbilledCost are summed. // The JPA values are kept as jpaUnbilledRevenue / jpaUnpostedCost for reference. // If estimatedvsactual is empty (not pushed yet) the JPA values are left in place. const EVA_COLLECTION = 'estimatedvsactual'; const toNum = v => { const n = typeof v === 'number' ? v : parseFloat(String(v ?? '').replace(/[,\s]/g, '')); return isNaN(n) ? 0 : n; }; const round2 = n => Math.round(n * 100) / 100; const evaDirection = lob => { const u = String(lob || '').toUpperCase(); return /EXP/.test(u) ? 'Export' : /IMP/.test(u) ? 'Import' : ''; }; function sumEvaByJob(evaRows) { const byJob = new Map(); for (const e of evaRows) { const job = String(e.jobNo ?? '').trim(); if (!job) continue; let t = byJob.get(job); if (!t) { t = { unbilledRevenue: 0, unbilledCost: 0, lines: 0, revLines: 0, costLines: 0, lob2: e.lob2 || '', direction: e.direction || evaDirection(e.lob2 || e.lob), etdEtaNew: e.etdEtaNew || '', etdEta: e.etdEta || '', ataAtd: e.ataAtd || '', jobDate: e.jobDate || '', location: e.location || '', jobOwner: e.jobOwner || '', operationLockDate: e.operationLockDate || '', organization: e.organization || '', salesPerson: e.salesPerson || '', sourceYear: e.sourceYear || '' }; byJob.set(job, t); } // USA naming first (unpostedCost), Canada sheet naming as fallback (unbilledCost) const rev = toNum(e.unbilledRevenue), cost = toNum(e.unpostedCost !== undefined && e.unpostedCost !== '' ? e.unpostedCost : e.unbilledCost); t.unbilledRevenue += rev; t.unbilledCost += cost; t.lines++; if (Math.abs(rev) > 0.005) t.revLines++; if (Math.abs(cost) > 0.005) t.costLines++; } return byJob; } // Writes the summed amounts onto the dashboard jobs, and returns the jobs that exist only in // estimatedvsactual (not in ca_ops_export / ca_ops_import) but have unbilled lines, so the // Revenue Not Billed / Cost Not Posted pages can still count them. function applyEvaAmounts(records, byJob) { if (!byJob.size) return { source: 'jpa', matched: 0, evaOnly: [] }; let matched = 0; const seen = new Set(); for (const r of records) { const id = String(r.shipmentId ?? '').trim(); const t = byJob.get(id); r.jpaUnbilledRevenue = r.unbilledRevenue; r.jpaUnpostedCost = r.unpostedCost; r.unbilledRevenue = t ? round2(t.unbilledRevenue) : 0; r.unpostedCost = t ? round2(t.unbilledCost) : 0; r.evaChargeLines = t ? t.lines : 0; r.evaUnbilledRevLines = t ? t.revLines : 0; r.evaUnbilledCostLines = t ? t.costLines : 0; if (t) { matched++; seen.add(id); } } const evaOnly = []; for (const [job, t] of byJob) { if (seen.has(job) || (!t.revLines && !t.costLines) || !t.direction) continue; evaOnly.push({ shipmentId: job, direction: t.direction, lob: t.lob2 || ('SEA ' + t.direction.toUpperCase()), jobDate: t.jobDate, periodDate: t.etdEta || t.jobDate, branch: t.location, location: t.location, customer: '', organization: t.organization, salesPerson: t.salesPerson, jobOwner: '', jobStatus: '', sourceYear: t.sourceYear, unbilledRevenue: round2(t.unbilledRevenue), unpostedCost: round2(t.unbilledCost), evaChargeLines: t.lines, evaUnbilledRevLines: t.revLines, evaUnbilledCostLines: t.costLines, evaOnly: true }); } return { source: EVA_COLLECTION, matched, evaOnly }; } // One summary per estimatedvsactual job, built only from estimatedvsactual (no JPA merge). // Used by the Revenue Not Billed / Cost Not Posted reports. function evaJobSummaries(byJob) { const out = []; for (const [job, t] of byJob) { if (!t.direction) continue; out.push({ shipmentId: job, direction: t.direction, lob: t.lob2 || ('SEA ' + t.direction.toUpperCase()), // Same as USA: period = ETD/ETA New (3-step fallback column), else ETD/ETA, else Job Date jobDate: t.jobDate, periodDate: t.etdEtaNew || t.etdEta || t.jobDate, etdEtaNew: t.etdEtaNew, etdEta: t.etdEta, ataAtd: t.ataAtd, branch: t.location || '(No Branch)', location: t.location, jobOwner: t.jobOwner || t.salesPerson, salesPerson: t.salesPerson, operationLockDate: t.operationLockDate, organization: t.organization, sourceYear: t.sourceYear, unbilledRevenue: round2(t.unbilledRevenue), unpostedCost: round2(t.unbilledCost), evaChargeLines: t.lines, evaUnbilledRevLines: t.revLines, evaUnbilledCostLines: t.costLines, evaJob: true }); } return out; } let cache = { payload: null, builtAt: null }; async function buildCache(db) { console.log('[CA-OPS] Building cache...'); const expDocs = await db.collection(COLS.Export).find({}).toArray(); const impDocs = await db.collection(COLS.Import).find({}).toArray(); const userDocs = await db.collection('users').find({}).toArray(); const evaRows = await db.collection(EVA_COLLECTION) .find({}, { projection: { _id: 0, jobNo: 1, unbilledRevenue: 1, unbilledCost: 1, unpostedCost: 1, lob: 1, lob2: 1, direction: 1, etdEtaNew: 1, etdEta: 1, ataAtd: 1, jobDate: 1, location: 1, jobOwner: 1, operationLockDate: 1, organization: 1, salesPerson: 1, sourceYear: 1 } }).toArray(); const all = [...expDocs, ...impDocs]; const evaByJob = sumEvaByJob(evaRows); const evaJobs = evaJobSummaries(evaByJob); const amounts = applyEvaAmounts(all, evaByJob); const records = all.map(compact); console.log(`[CA-OPS] Export: ${expDocs.length}, Import: ${impDocs.length}, users: ${userDocs.length}, ` + `amounts from ${amounts.source} (${evaRows.length} charge lines, ${amounts.matched} jobs matched, ${amounts.evaOnly.length} only in ${EVA_COLLECTION})`); cache = { payload: { records, users: userDocs, amountsSource: amounts.source, evaJobsMatched: amounts.matched, evaOnlyRecords: amounts.evaOnly.map(compact), evaJobs: evaJobs.map(compact) }, builtAt: new Date().toISOString() }; return cache; } export default async function handler(req, res) { res.setHeader('Access-Control-Allow-Origin', '*'); res.setHeader('Access-Control-Allow-Methods', 'POST, GET, OPTIONS'); res.setHeader('Access-Control-Allow-Headers', 'Content-Type, x-batch-secret'); if (req.method === 'OPTIONS') return res.status(200).end(); if (req.method === 'POST') { const secret = req.headers['x-batch-secret']; if (!SECRET || secret !== SECRET) return res.status(401).json({ error: 'Unauthorized' }); const { action, records, direction, collection } = req.body || {}; const colName = collection ? (EXTRA_COLLECTIONS.has(collection) ? collection : null) : COLS[direction]; if (!colName) return res.status(400).json({ error: collection ? 'Collection not allowed: ' + collection : 'Invalid direction' }); try { const db = (await getClient()).db(DB); if (action === 'wipe') { const result = await db.collection(colName).deleteMany({}); cache = { payload: null, builtAt: null }; return res.status(200).json({ deleted: result.deletedCount, collection: colName }); } if (action === 'push') { if (!records || !records.length) return res.status(400).json({ error: 'No records' }); const result = await db.collection(colName).insertMany(records, { ordered: false }); cache = { payload: null, builtAt: null }; return res.status(200).json({ inserted: result.insertedCount, collection: colName }); } return res.status(400).json({ error: 'Unknown action' }); } catch (e) { console.error(e); return res.status(500).json({ error: e.message }); } } if (req.method === 'GET') { if (req.query && req.query.ping) return res.status(200).json({ ok: true }); try { const force = req.query && req.query.forceRefresh === '1'; if (force || !cache.payload || !cache.builtAt || (Date.now() - new Date(cache.builtAt)) > TTL_MS) { await buildCache((await getClient()).db(DB)); res.setHeader('X-Cache', 'MISS'); } else { res.setHeader('X-Cache', 'HIT'); } res.setHeader('Cache-Control', 'no-store'); res.setHeader('X-Cache-Age', Math.floor((Date.now() - new Date(cache.builtAt)) / 1000)); return res.status(200).json({ ...cache.payload, builtAt: cache.builtAt }); } catch (e) { console.error(e); return res.status(500).json({ error: e.message }); } } return res.status(405).json({ error: 'Method not allowed' }); }
